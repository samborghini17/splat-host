# --- KONFIGURATION ---
$sourceBase = "D:\ICL-Scans\KIO_splats_export"
$targetBase = "D:\ICL-Scans\KIO_splats_playcanvas_lod"

# Speicherlimit für Node.js setzen
$env:NODE_OPTIONS = "--max-old-space-size=16000"

# Funktion zur GitHub-freundlichen Formatierung von Ordnernamen
function Get-GitHubFriendlyName($name) {
    $name = $name.ToLower()
    $name = $name -replace 'ä', 'ae' -replace 'ö', 'oe' -replace 'ü', 'ue' -replace 'ß', 'ss'
    $name = $name -replace '[^a-z0-9]+', '-'
    $name = $name.Trim('-')
    return $name
}

# Erstelle Ziel-Hauptverzeichnis, falls nicht vorhanden
if (!(Test-Path $targetBase)) {
    New-Item -ItemType Directory -Force -Path $targetBase | Out-Null
}

# Alle Unterordner im Export-Verzeichnis holen
$folders = Get-ChildItem -Path $sourceBase -Directory

foreach ($folder in $folders) {
    $cleanName = Get-GitHubFriendlyName $folder.Name
    $targetDir = Join-Path -Path $targetBase -ChildPath $cleanName
    
    # Zielordner erstellen
    if (!(Test-Path $targetDir)) {
        New-Item -ItemType Directory -Force -Path $targetDir | Out-Null
    }

    Write-Host "`n=======================================================" -ForegroundColor Cyan
    Write-Host "Verarbeite Ordner: $($folder.Name) -> $cleanName" -ForegroundColor Cyan
    Write-Host "=======================================================" -ForegroundColor Cyan

    # In den Zielordner wechseln, damit lod-meta.json und WebPs dort generiert werden
    Push-Location $targetDir

    # --- 1. DYNAMISCHE LOD TRANSFORMATION ---
    $sourcePly0 = Join-Path -Path $folder.FullName -ChildPath "point_cloud.ply"

    if (Test-Path $sourcePly0) {
        Write-Host "Starte dynamische LOD-Transformation..." -ForegroundColor Yellow

        # Basis-Command mit absolutem Pfad zur LOD 0 Datei
        $cmdLOD = "splat-transform `"$sourcePly0`" -l 0"

        # Dynamisch alle weiteren Stufen (1, 2, 3...) aus dem Quellordner suchen und anfügen
        $lodIndex = 1
        while ($true) {
            $nextPly = Join-Path -Path $folder.FullName -ChildPath "point_cloud_$lodIndex.ply"
            
            if (Test-Path $nextPly) {
                Write-Host "LOD $lodIndex gefunden: point_cloud_$lodIndex.ply" -ForegroundColor Gray
                # Füge die nächste Datei mit absolutem Pfad zum Command hinzu
                $cmdLOD += " `"$nextPly`" -l $lodIndex"
                $lodIndex++
            } else {
                # Sobald keine weitere point_cloud_X.ply gefunden wird, breche die Schleife ab
                break
            }
        }

        # Output-Datei und Filter als Abschluss des Commands anfügen
        $cmdLOD += " -w lod-meta.json --filter-nan --filter-harmonics 0"

        Write-Host "> $cmdLOD" -ForegroundColor DarkGray
        Invoke-Expression $cmdLOD
    }
    else {
        Write-Host "Überspringe LOD: Keine point_cloud.ply im Quellordner gefunden." -ForegroundColor Red
    }

    # --- 2. ENVIRONMENT TRANSFORMATION ---
    $envPly = Join-Path -Path $folder.FullName -ChildPath "environment.ply"
    
    if (Test-Path $envPly) {
        Write-Host "`nStarte Transformation für environment.ply..." -ForegroundColor Yellow
        
        $cmdEnv = "splat-transform `"$envPly`" -w environment.sog --filter-nan --filter-harmonics 0"
        
        Write-Host "> $cmdEnv" -ForegroundColor DarkGray
        Invoke-Expression $cmdEnv
    }
    else {
        Write-Host "Überspringe Environment: Keine environment.ply gefunden." -ForegroundColor DarkGray
    }

    # Zurück ins Hauptverzeichnis für den nächsten Ordner-Durchlauf
    Pop-Location
}

Write-Host "`nFertig! Alle Ordner wurden nacheinander und mit dynamischem LOD verarbeitet." -ForegroundColor Green
